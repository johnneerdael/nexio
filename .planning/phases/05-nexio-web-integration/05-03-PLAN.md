import { createError } from 'h3'
import { bearerToken, okJson, readJsonBody, supabaseUser } from '~/server/utils/supabase'
import { buildTraktHeaders, readTraktErrorDetails } from '~/server/utils/trakt'

type DeviceCodeBody = { profileIndex?: number }

export default defineEventHandler(async (event) => {
  const body = await readJsonBody<DeviceCodeBody>(event)
  bearerToken(event)
  await supabaseUser(event)

  const profileIndex = body.profileIndex
  if (!profileIndex || profileIndex < 1 || profileIndex > 4) {
    throw createError({ statusCode: 400, statusMessage: 'profileIndex must be 1-4.' })
  }

  const config = useRuntimeConfig()
  const clientId = String(config.traktClientId || '').trim()
  if (!clientId) {
    throw createError({ statusCode: 503, statusMessage: 'TRAKT_CLIENT_ID is not configured.' })
  }

  // Same Trakt API call as account-level route
  const upstream = await fetch('https://api.trakt.tv/oauth/device/code', {
    method: 'POST',
    headers: buildTraktHeaders({ clientId }),
    body: JSON.stringify({ client_id: clientId })
  })

  // Same error handling as account-level route (check for Cloudflare blocks, etc.)
  if (!upstream.ok) {
    const details = await readTraktErrorDetails(upstream)
    if (details.blockedByCloudflare) {
      throw createError({ statusCode: 502, statusMessage: `Trakt is temporarily unreachable.${details.cfRay ? ` CF Ray: ${details.cfRay}` : ''}` })
    }
    throw createError({ statusCode: details.status, statusMessage: details.responseText })
  }

  const payload = await upstream.json()
  return okJson({
    deviceCode: payload.device_code,
    userCode: payload.user_code,
    verificationUrl: payload.verification_url,
    verificationUrlComplete: payload.verification_url_complete,
    expiresIn: payload.expires_in,
    interval: payload.interval,
    profileIndex,
    startedAt: new Date().toISOString()
  })
})

**`server/api/integrations/profiles/trakt/device-token.post.ts`**
Similar to account-level but stores tokens via `service_set_profile_auth_token` RPC with `p_profile_index`:
```typescript
import { createError } from 'h3'
import { bearerToken, okJson, readJsonBody, supabaseFetch, supabaseUser } from '~/server/utils/supabase'
import { buildTraktHeaders, readTraktErrorDetails } from '~/server/utils/trakt'

type DeviceTokenBody = { deviceCode?: string; profileIndex?: number }

export default defineEventHandler(async (event) => {
  bearerToken(event)
  const user = await supabaseUser(event)
  const body = await readJsonBody<DeviceTokenBody>(event)

  const deviceCode = body.deviceCode?.trim()
  const profileIndex = body.profileIndex
  if (!deviceCode) throw createError({ statusCode: 400, statusMessage: 'Trakt device code is required.' })
  if (!profileIndex || profileIndex < 1 || profileIndex > 4) throw createError({ statusCode: 400, statusMessage: 'profileIndex must be 1-4.' })

  const config = useRuntimeConfig()
  const clientId = String(config.traktClientId || '').trim()
  const clientSecret = String(config.traktClientSecret || '').trim()
  if (!clientId || !clientSecret) throw createError({ statusCode: 503, statusMessage: 'TRAKT credentials are not configured.' })

  const tokenResponse = await fetch('https://api.trakt.tv/oauth/device/token', {
    method: 'POST',
    headers: buildTraktHeaders({ clientId }),
    body: JSON.stringify({ code: deviceCode, client_id: clientId, client_secret: clientSecret })
  })

  if (!tokenResponse.ok) {
    const details = await readTraktErrorDetails(tokenResponse)
    if (details.blockedByCloudflare) throw createError({ statusCode: 502, statusMessage: 'Trakt temporarily unreachable.' })
    return okJson({ status: details.status, pending: details.status === 400 || details.status === 429, approved: false })
  }

  const tokenPayload = await tokenResponse.json()

  // Fetch Trakt username for display
  const userSettingsResponse = await fetch('https://api.trakt.tv/users/settings', {
    headers: buildTraktHeaders({ clientId, accessToken: tokenPayload.access_token })
  })
  const userSettings = userSettingsResponse.ok ? await userSettingsResponse.json() : { user: {} }

  // Store access token scoped to profile — use service_set_profile_auth_token RPC
  await supabaseFetch('/rest/v1/rpc/service_set_profile_auth_token', {
    method: 'POST',
    body: JSON.stringify({
      p_user_id: user.id,
      p_profile_index: profileIndex,
      p_token_type: 'trakt_access_token',
      p_token_payload: {
        accessToken: tokenPayload.access_token,
        tokenType: tokenPayload.token_type,
        createdAt: tokenPayload.created_at,
        expiresIn: tokenPayload.expires_in
      },
      p_masked_preview: `Connected ••••${tokenPayload.access_token.slice(-4)}`,
      p_source: 'web-trakt'
    })
  }, undefined, true) // useServiceRole=true

  // Store refresh token scoped to profile
  await supabaseFetch('/rest/v1/rpc/service_set_profile_auth_token', {
    method: 'POST',
    body: JSON.stringify({
      p_user_id: user.id,
      p_profile_index: profileIndex,
      p_token_type: 'trakt_refresh_token',
      p_token_payload: { refreshToken: tokenPayload.refresh_token },
      p_masked_preview: `Stored ••••${tokenPayload.refresh_token.slice(-4)}`,
      p_source: 'web-trakt'
    })
  }, undefined, true)

  return okJson({
    status: 200,
    pending: false,
    approved: true,
    profileIndex,
    traktAuth: {
      connected: true,
      username: userSettings.user?.username ?? '',
      userSlug: userSettings.user?.ids?.slug ?? '',
      connectedAt: new Date().toISOString(),
      pending: false
    }
  })
})
```

**`server/api/integrations/profiles/trakt/disconnect.post.ts`**
Per D-12: MUST call Trakt revoke endpoint before deleting tokens. The existing account-level `disconnectTrakt` in usePortalStore does NOT call revoke — this route does:
```typescript
import { createError } from 'h3'
import { bearerToken, okJson, readJsonBody, supabaseFetch, supabaseUser } from '~/server/utils/supabase'
import { buildTraktHeaders } from '~/server/utils/trakt'

type DisconnectBody = { profileIndex?: number }

export default defineEventHandler(async (event) => {
  bearerToken(event)
  const user = await supabaseUser(event)
  const body = await readJsonBody<DisconnectBody>(event)
  const profileIndex = body.profileIndex
  if (!profileIndex || profileIndex < 1 || profileIndex > 4) throw createError({ statusCode: 400, statusMessage: 'profileIndex must be 1-4.' })

  const config = useRuntimeConfig()
  const clientId = String(config.traktClientId || '').trim()
  const clientSecret = String(config.traktClientSecret || '').trim()

  // Step 1: Read the existing access token from Supabase to revoke it
  // Query profile_auth_tokens directly via Postgrest (RLS ensures user_id match)
  const tokens = await supabaseFetch<Array<{ token_payload: { accessToken?: string } }>>(
    `/rest/v1/profile_auth_tokens?user_id=eq.${user.id}&profile_index=eq.${profileIndex}&token_type=eq.trakt_access_token&select=token_payload`,
    { method: 'GET' },
    undefined,
    true // service role to read token_payload
  )

  const accessToken = tokens?.[0]?.token_payload?.accessToken
  if (accessToken && clientId && clientSecret) {
    // Step 2: Call Trakt revoke endpoint (D-12 — full revoke)
    try {
      await fetch('https://api.trakt.tv/oauth/revoke', {
        method: 'POST',
        headers: buildTraktHeaders({ clientId }),
        body: JSON.stringify({
          token: accessToken,
          client_id: clientId,
          client_secret: clientSecret
        })
      })
    } catch (e) {
      // Log but don't fail — revoke is best-effort; deletion must still happen
      console.error('[trakt][profile-disconnect] revoke failed:', e)
    }
  }

  // Step 3: Delete all Trakt tokens for this profile from Supabase
  await supabaseFetch('/rest/v1/rpc/service_delete_profile_auth_tokens', {
    method: 'POST',
    body: JSON.stringify({
      p_user_id: user.id,
      p_profile_index: profileIndex,
      p_token_type_prefix: 'trakt'
    })
  }, undefined, true)

  return okJson({ disconnected: true, profileIndex, provider: 'trakt' })
})
```
