```sql
CREATE TABLE IF NOT EXISTS public.profile_auth_tokens (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  profile_index INT NOT NULL CHECK (profile_index BETWEEN 1 AND 4),
  token_type TEXT NOT NULL CHECK (token_type IN ('trakt_access_token', 'trakt_refresh_token', 'simkl_access_token')),
  token_payload JSONB NOT NULL DEFAULT '{}'::jsonb,
  masked_preview TEXT DEFAULT '',
  source TEXT DEFAULT 'web',
  linked BOOLEAN NOT NULL DEFAULT true,
  revoked_at TIMESTAMPTZ DEFAULT NULL,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  UNIQUE (user_id, profile_index, token_type)
);
```

Create a service-role RPC for setting per-profile tokens (used by server routes with useServiceRole=true):
```sql
CREATE OR REPLACE FUNCTION public.service_set_profile_auth_token(
  p_user_id UUID,
  p_profile_index INT,
  p_token_type TEXT,
  p_token_payload JSONB,
  p_masked_preview TEXT DEFAULT '',
  p_source TEXT DEFAULT 'web'
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
  INSERT INTO public.profile_auth_tokens (user_id, profile_index, token_type, token_payload, masked_preview, source, linked, revoked_at, updated_at)
  VALUES (p_user_id, p_profile_index, p_token_type, p_token_payload, p_masked_preview, p_source, true, NULL, now())
  ON CONFLICT (user_id, profile_index, token_type)
  DO UPDATE SET
    token_payload = EXCLUDED.token_payload,
    masked_preview = EXCLUDED.masked_preview,
    source = EXCLUDED.source,
    linked = true,
    revoked_at = NULL,
    updated_at = now();
END;
$$;
```

Create profile CRUD RPCs. Profile upsert (per D-01, D-07) MUST make preserve-vs-clear unambiguous: `p_avatar_url = NULL` preserves the existing avatar, while `p_clear_avatar = TRUE` clears `avatar_url` to NULL. This explicit contract is required by WEB-05 photo deletion.
```sql
CREATE OR REPLACE FUNCTION public.profile_upsert(
  p_profile_index INT,
  p_name TEXT,
  p_avatar_color_hex TEXT DEFAULT NULL,
  p_avatar_url TEXT DEFAULT NULL,
  p_clear_avatar BOOLEAN DEFAULT FALSE
)
RETURNS SETOF public.profiles
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_user_id UUID := auth.uid();
  v_profile_count INT;
BEGIN
  IF p_profile_index < 1 OR p_profile_index > 4 THEN
    RAISE EXCEPTION 'profile_index must be between 1 and 4';
  END IF;

  IF length(trim(p_name)) < 1 OR length(trim(p_name)) > 30 THEN
    RAISE EXCEPTION 'Profile name must be 1-30 characters';
  END IF;

  -- Check max 4 profiles per user (INFRA-02)
  SELECT count(*) INTO v_profile_count
  FROM public.profiles
  WHERE user_id = v_user_id AND profile_index != p_profile_index;

  IF v_profile_count >= 4 THEN
    RAISE EXCEPTION 'Maximum 4 profiles per account';
  END IF;

  RETURN QUERY
  INSERT INTO public.profiles (user_id, profile_index, name, avatar_color_hex, avatar_url, updated_at)
  VALUES (v_user_id, p_profile_index, trim(p_name), COALESCE(p_avatar_color_hex, '#1E88E5'), p_avatar_url, now())
  ON CONFLICT (user_id, profile_index)
  DO UPDATE SET
    name = EXCLUDED.name,
    avatar_color_hex = COALESCE(EXCLUDED.avatar_color_hex, profiles.avatar_color_hex),
    avatar_url = CASE
      WHEN p_clear_avatar THEN NULL
      WHEN p_avatar_url IS NOT NULL THEN p_avatar_url
      ELSE profiles.avatar_url
    END,
    updated_at = now()
  RETURNING *;
END;
$$;
```

Profile delete (refuses profile_index = 1 per security requirement):
```sql
CREATE OR REPLACE FUNCTION public.profile_delete(
  p_profile_index INT
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_user_id UUID := auth.uid();
BEGIN
  IF p_profile_index = 1 THEN
    RAISE EXCEPTION 'Cannot delete the primary profile';
  END IF;

  IF p_profile_index < 2 OR p_profile_index > 4 THEN
    RAISE EXCEPTION 'Invalid profile index';
  END IF;

  -- Delete auth tokens for this profile
  DELETE FROM public.profile_auth_tokens
  WHERE user_id = v_user_id AND profile_index = p_profile_index;

  -- Preserve Phase 4 profile-settings cleanup behavior if that table exists.
  -- This keeps web profile deletion from leaving v8 profile_settings blobs behind.
  IF to_regclass('public.profile_settings') IS NOT NULL THEN
    EXECUTE 'DELETE FROM public.profile_settings WHERE user_id = $1 AND profile_id = $2'
    USING v_user_id, p_profile_index;
  END IF;

  -- Delete the public avatar object at profile-avatars/{userId}/{profileIndex}.jpg.
  -- The storage.objects name is the bucket-relative path: {userId}/{profileIndex}.jpg.
  DELETE FROM storage.objects
  WHERE bucket_id = 'profile-avatars'
    AND name = v_user_id::text || '/' || p_profile_index::text || '.jpg';

  -- Delete the profile
  DELETE FROM public.profiles
  WHERE user_id = v_user_id AND profile_index = p_profile_index;
END;
$$;
```

**Create `nexio-web/server/api/account/profiles/upsert.post.ts`:**
Create or update a profile (name, avatar color, avatar URL). Calls the profile_upsert RPC.

```typescript
import { createError } from 'h3'
import { bearerToken, okJson, readJsonBody, supabaseFetch, supabaseUser } from '~/server/utils/supabase'

type UpsertBody = {
  profileIndex: number
  name: string
  avatarColorHex?: string
  avatarUrl?: string | null
}

export default defineEventHandler(async (event) => {
  const body = await readJsonBody<UpsertBody>(event)
  const token = bearerToken(event)
  await supabaseUser(event)

  if (!body.profileIndex || body.profileIndex < 1 || body.profileIndex > 4) {
    throw createError({ statusCode: 400, statusMessage: 'profileIndex must be 1-4.' })
  }

  const name = (body.name || '').trim()
  if (name.length < 1 || name.length > 30) {
    throw createError({ statusCode: 400, statusMessage: 'Profile name must be 1-30 characters.' })
  }

  const result = await supabaseFetch<unknown[]>(
    '/rest/v1/rpc/profile_upsert',
    {
      method: 'POST',
      body: JSON.stringify({
        p_profile_index: body.profileIndex,
        p_name: name,
        p_avatar_color_hex: body.avatarColorHex || null,
        p_avatar_url: body.avatarUrl !== undefined ? body.avatarUrl : null,
        p_clear_avatar: false
      })
    },
    token
  )

  return okJson(Array.isArray(result) ? result[0] : result)
})
```

**Create `nexio-web/server/api/account/profiles/delete.post.ts`:**
Delete a non-primary profile. Client-side AND server-side guard against profile_index=1.

```typescript
import { createError } from 'h3'
import { bearerToken, okJson, readJsonBody, supabaseFetch, supabaseUser } from '~/server/utils/supabase'

type DeleteBody = {
  profileIndex: number
}

export default defineEventHandler(async (event) => {
  const body = await readJsonBody<DeleteBody>(event)
  const token = bearerToken(event)
  await supabaseUser(event)

  if (!body.profileIndex || body.profileIndex < 2 || body.profileIndex > 4) {
    throw createError({ statusCode: 400, statusMessage: 'Cannot delete primary profile or invalid index.' })
  }

  await supabaseFetch<void>(
    '/rest/v1/rpc/profile_delete',
    {
```
