**1. Create `nexio-web/server/api/account/profiles/settings.get.ts`**

Loads the per-profile settings blob from Supabase using the resolved Phase 4 RPC name `sync_pull_profile_settings_blob`. Do not use the old non-blob pull RPC, and do not return empty settings as a fallback if the RPC fails.

```typescript
import { createError } from 'h3'
import { bearerToken, okJson, supabaseFetch, supabaseUser } from '~/server/utils/supabase'

export default defineEventHandler(async (event) => {
  const token = bearerToken(event)
  await supabaseUser(event)

  const query = getQuery(event)
  const profileIndex = Number(query.profileIndex)
  if (!profileIndex || profileIndex < 1 || profileIndex > 4) {
    throw createError({ statusCode: 400, statusMessage: 'profileIndex query param must be 1-4.' })
  }

  const result = await supabaseFetch<{ settings_json?: string; updated_at?: string }>(
    '/rest/v1/rpc/sync_pull_profile_settings_blob',
    {
      method: 'POST',
      body: JSON.stringify({
        p_profile_id: profileIndex,
        p_platform: 'web'
      })
    },
    token
  )

  const settings = result?.settings_json ? JSON.parse(result.settings_json) : {}
  return okJson({
    settings,
    updatedAt: result?.updated_at ?? null,
    profileIndex
  })
})
```

**2. Create `nexio-web/server/api/account/profiles/settings.post.ts`**

Saves the per-profile settings blob to Supabase using the resolved Phase 4 RPC name `sync_push_profile_settings_blob`.

```typescript
import { createError } from 'h3'
import { bearerToken, okJson, readJsonBody, supabaseFetch, supabaseUser } from '~/server/utils/supabase'

type ProfileSettingsBody = {
  profileIndex: number
  settings: Record<string, unknown>
  changedPaths?: string[]
}

export default defineEventHandler(async (event) => {
  const body = await readJsonBody<ProfileSettingsBody>(event)
  const token = bearerToken(event)
  await supabaseUser(event)

  if (!body.profileIndex || body.profileIndex < 1 || body.profileIndex > 4) {
    throw createError({ statusCode: 400, statusMessage: 'profileIndex must be 1-4.' })
  }

  if (!body.settings || typeof body.settings !== 'object') {
    throw createError({ statusCode: 400, statusMessage: 'settings payload is required.' })
  }

  const result = await supabaseFetch<{ updated_at?: string }>(
    '/rest/v1/rpc/sync_push_profile_settings_blob',
    {
      method: 'POST',
      body: JSON.stringify({
        p_profile_id: body.profileIndex,
        p_settings_json: JSON.stringify(body.settings),
        p_platform: 'web'
      })
    },
    token
  )

  return okJson({
    updatedAt: result?.updated_at ?? new Date().toISOString(),
    profileIndex: body.profileIndex
  })
})
```

    state.value.profileSettings[profileIndex] = {

      loading: true,

      settings: state.value.profileSettings[profileIndex]?.settings || {},
