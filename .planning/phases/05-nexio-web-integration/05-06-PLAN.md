**1. Create `app/src/main/java/com/nexio/tv/core/sync/ProfileWebSyncService.kt`**

A service that pulls profile metadata and per-profile auth tokens from Supabase. Called on app launch (D-04) and on profile switch. Uses last-write-wins (D-05, D-11) for conflict resolution.

```kotlin
package com.nexio.tv.core.sync

import io.github.jan.supabase.postgrest.Postgrest
import io.github.jan.supabase.postgrest.query.Columns
import kotlinx.serialization.SerialName
import kotlinx.serialization.Serializable
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Per-profile auth token stored in Supabase profile_auth_tokens table.
 * Used for syncing Trakt/Simkl tokens linked via nexio-web.
 */
@Serializable
data class ProfileAuthToken(
    @SerialName("profile_index") val profileIndex: Int,
    @SerialName("token_type") val tokenType: String,
    @SerialName("token_payload") val tokenPayload: Map<String, String> = emptyMap(),
    @SerialName("masked_preview") val maskedPreview: String = "",
    val source: String = "",
    @SerialName("updated_at") val updatedAt: String = ""
)

/**
 * Pulls profile metadata and per-profile auth tokens from Supabase.
 *
 * Called:
 * - On app launch (D-04: next-launch sync)
 * - On profile switch
 *
 * Conflict resolution: last-write-wins by updated_at timestamp (D-05, D-11).
 */
@Singleton
class ProfileWebSyncService @Inject constructor(
    private val postgrest: Postgrest
) {

    /**
     * Pull all profiles for the current user from Supabase.
     * Returns the list from the profiles table, ordered by profile_index.
     */
    suspend fun pullProfiles(): List<com.nexio.tv.data.remote.supabase.SupabaseProfile> {
        return try {
            postgrest.from("profiles")
                .select()
                .decodeList()
        } catch (e: Exception) {
            emptyList()
        }
    }

    /**
     * Pull per-profile auth tokens for a specific profile.
     * Returns tokens from profile_auth_tokens table scoped to the profile_index.
     * RLS ensures only the authenticated user's tokens are returned.
     */
    suspend fun pullProfileAuthTokens(profileIndex: Int): List<ProfileAuthToken> {
        return try {
            postgrest.from("profile_auth_tokens")
                .select() {
                    filter {
                        eq("profile_index", profileIndex)
                    }
                }
                .decodeList()
        } catch (e: Exception) {
            emptyList()
        }
    }

    /**
     * Sync per-profile Trakt tokens to the local DataStore.
     *
     * Uses most-recent-wins (D-11): compares remote updated_at with local
     * last_linked_at. If remote is newer, overwrites local DataStore values.
     * If local is newer, keeps local (TV-side OAuth takes precedence).
     *
     * @param profileIndex the profile to sync
     * @param localLastLinkedAt ISO timestamp of when the local tokens were last set, or null if never
     * @param onTokenUpdate callback invoked with (accessToken, refreshToken) when remote tokens are newer
     */
    suspend fun syncProfileTraktTokens(
        profileIndex: Int,
        localLastLinkedAt: String?,
        onTokenUpdate: suspend (accessToken: String, refreshToken: String?) -> Unit
    ) {
        val remoteTokens = pullProfileAuthTokens(profileIndex)
        val accessTokenEntry = remoteTokens.find { it.tokenType == "trakt_access_token" }
        val refreshTokenEntry = remoteTokens.find { it.tokenType == "trakt_refresh_token" }

        if (accessTokenEntry == null) return // No remote Trakt tokens for this profile

        // Most-recent-wins: compare timestamps (D-11)
        val remoteUpdatedAt = accessTokenEntry.updatedAt
        if (localLastLinkedAt != null && localLastLinkedAt >= remoteUpdatedAt) {
            return // Local is newer or same — keep local
        }

        // Remote is newer — update local DataStore
        val accessToken = accessTokenEntry.tokenPayload["accessToken"] ?: return
        val refreshToken = refreshTokenEntry?.tokenPayload["refreshToken"]
        onTokenUpdate(accessToken, refreshToken)
    }

    /**
     * Same as syncProfileTraktTokens but for Simkl (access token only, no refresh token).
     */
    suspend fun syncProfileSimklTokens(
        profileIndex: Int,
        localLastLinkedAt: String?,
        onTokenUpdate: suspend (accessToken: String) -> Unit
    ) {
        val remoteTokens = pullProfileAuthTokens(profileIndex)
        val accessTokenEntry = remoteTokens.find { it.tokenType == "simkl_access_token" }

        if (accessTokenEntry == null) return

        val remoteUpdatedAt = accessTokenEntry.updatedAt
        if (localLastLinkedAt != null && localLastLinkedAt >= remoteUpdatedAt) {
            return
        }

        val accessToken = accessTokenEntry.tokenPayload["accessToken"] ?: return
        onTokenUpdate(accessToken)
    }
}
```

The service is a `@Singleton` injected via Hilt. It uses the Postgrest client already provided by SupabaseModule. The caller (typically the existing startup flow or profile switch handler) invokes `pullProfiles()` to get the latest profile list and `syncProfileTraktTokens`/`syncProfileSimklTokens` to update local DataStore values when remote tokens are newer.

**Important:** The `onTokenUpdate` callback pattern allows the caller to write to the appropriate per-profile DataStore without this service needing to know the DataStore implementation details. This keeps domain code free of Android framework dependencies per CLAUDE.md.

**2. Create `app/src/main/java/com/nexio/tv/ui/components/ProfileAvatarImage.kt`**

A reusable composable for displaying a profile avatar with photo/color fallback (D-16) and Coil URL-based cache invalidation (D-17).

```kotlin
package com.nexio.tv.ui.components

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.size
import androidx.compose.foundation.shape.CircleShape
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.draw.clip
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.layout.ContentScale
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.Dp
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import coil.compose.AsyncImage
import coil.request.ImageRequest

/**
 * Displays a profile avatar image from URL, or falls back to a colored circle
 * with the profile's first initial letter.
 *
 * Uses Coil AsyncImage with URL-based cache key (D-17): when avatarUrl changes
 * in the sync payload, Coil automatically re-fetches because the URL string
 * (including the ?t= cache-bust query param) is different.
 *
 * Falls back to colored circle with initial when avatarUrl is null (D-16).
 *
 * @param avatarUrl Public Supabase Storage URL or null
 * @param avatarColorHex Hex color string for fallback circle (e.g. "#1E88E5")
 * @param name Profile name (first character used for fallback initial)
 * @param size Size of the avatar circle in dp
 * @param modifier Optional modifier
 */
@Composable
fun ProfileAvatarImage(
    avatarUrl: String?,
    avatarColorHex: String,
    name: String,
    size: Dp = 48.dp,
    modifier: Modifier = Modifier
) {
    if (avatarUrl != null) {
        AsyncImage(
            model = ImageRequest.Builder(LocalContext.current)
                .data(avatarUrl)
                .crossfade(true)
                .build(),
            contentDescription = name,
            contentScale = ContentScale.Crop,
            modifier = modifier
                .size(size)
                .clip(CircleShape)
        )
    } else {
        // Color circle fallback with initial letter
        val color = try {
            Color(android.graphics.Color.parseColor(avatarColorHex))
        } catch (e: Exception) {
            Color(0xFF1E88E5) // Default blue
        }

        Box(
            contentAlignment = Alignment.Center,
            modifier = modifier
                .size(size)
                .clip(CircleShape)
                .background(color)
        ) {
            Text(
                text = name.firstOrNull()?.uppercase() ?: "?",
                color = Color.White,
                fontWeight = FontWeight.Bold,
                fontSize = (size.value * 0.4f).sp
            )
        }
    }
}
```

This composable replaces any existing inline avatar rendering. Callers should use `ProfileAvatarImage` instead of manually checking for avatarUrl and rendering either AsyncImage or a Box. This is used by:

- Settings header (existing, from Phase 3)
