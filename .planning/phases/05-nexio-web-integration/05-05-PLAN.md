**Create `nexio-web/server/api/account/profiles/photo.post.ts`**

This route accepts a multipart file upload, validates the MIME type, resizes to 256x256 using sharp, uploads to Supabase Storage, and updates the profile's avatar_url.

```typescript
import { createError, readMultipartFormData } from 'h3'
import sharp from 'sharp'
import { bearerToken, okJson, supabaseFetch, supabaseUser, getSupabaseConfig } from '~/server/utils/supabase'

const ALLOWED_MIME_TYPES = ['image/jpeg', 'image/png', 'image/webp']
const MAX_FILE_SIZE = 10 * 1024 * 1024 // 10 MB before resize
const TARGET_SIZE = 256
const BUCKET = 'profile-avatars'

export default defineEventHandler(async (event) => {
  const token = bearerToken(event)
  const user = await supabaseUser(event)

  const parts = await readMultipartFormData(event)
  const filePart = parts?.find(p => p.name === 'photo')

  if (!filePart?.data) {
    throw createError({ statusCode: 400, statusMessage: 'No file provided.' })
  }

  // Read profileIndex from form data
  const profileIndexPart = parts?.find(p => p.name === 'profileIndex')
  const profileIndex = profileIndexPart ? Number(new TextDecoder().decode(profileIndexPart.data)) : 0
  if (!profileIndex || profileIndex < 1 || profileIndex > 4) {
    throw createError({ statusCode: 400, statusMessage: 'profileIndex must be 1-4.' })
  }

  // Validate MIME type (T-05-03 mitigation: malicious file upload)
  const mimeType = filePart.type || ''
  if (!ALLOWED_MIME_TYPES.includes(mimeType)) {
    throw createError({ statusCode: 400, statusMessage: `File type ${mimeType} not allowed. Use JPEG, PNG, or WebP.` })
  }

  // Validate file size before processing
  if (filePart.data.length > MAX_FILE_SIZE) {
    throw createError({ statusCode: 400, statusMessage: 'File too large. Maximum 10 MB.' })
  }

  // Resize with sharp (D-15: server-side resize to 256x256)
  // sharp validates that the input is actually an image — non-images throw an error
  let resizedBuffer: Buffer
  try {
    resizedBuffer = await sharp(filePart.data)
      .resize(TARGET_SIZE, TARGET_SIZE, { fit: 'cover' })
      .jpeg({ quality: 85 })
      .toBuffer()
  } catch (e) {
    throw createError({ statusCode: 400, statusMessage: 'Invalid image file. Could not process.' })
  }

  // Upload to Supabase Storage (D-14: public bucket, namespaced by userId/profileIndex)
  // Path: profile-avatars/{userId}/{profileIndex}.jpg
  const storagePath = `${user.id}/${profileIndex}.jpg`
  const config = getSupabaseConfig()

  // Use service role for storage upload
  const uploadUrl = `${config.url}/storage/v1/object/${BUCKET}/${storagePath}`
  const uploadResponse = await fetch(uploadUrl, {
    method: 'PUT',
    headers: {
      'Authorization': `Bearer ${config.serviceRoleKey}`,
      'Content-Type': 'image/jpeg',
      'x-upsert': 'true' // Overwrite existing file
    },
    body: resizedBuffer
  })

  if (!uploadResponse.ok) {
    const errorText = await uploadResponse.text()
    console.error('[photo.post] Storage upload failed:', errorText)
    throw createError({ statusCode: 500, statusMessage: 'Failed to upload photo to storage.' })
  }

  // Build the public URL
  const avatarUrl = `${config.url}/storage/v1/object/public/${BUCKET}/${storagePath}?t=${Date.now()}`

  // Update the profile's avatar_url via the profile_upsert RPC
  // We need to fetch the current profile name to pass it (RPC requires name)
  const profiles = await supabaseFetch<Array<{ name: string; avatar_color_hex: string }>>(
    `/rest/v1/profiles?profile_index=eq.${profileIndex}&select=name,avatar_color_hex`,
    { method: 'GET' },
    token
  )

  const currentProfile = profiles?.[0]
  if (currentProfile) {
    await supabaseFetch(
      '/rest/v1/rpc/profile_upsert',
      {
        method: 'POST',
        body: JSON.stringify({
          p_profile_index: profileIndex,
          p_name: currentProfile.name,
          p_avatar_color_hex: currentProfile.avatar_color_hex,
          p_avatar_url: avatarUrl,
          p_clear_avatar: false
        })
      },
      token
    )
  }

  return okJson({ avatarUrl, profileIndex })
})
```

**Create `nexio-web/server/api/account/profiles/photo.delete.ts`**

Removes the avatar photo from Storage and clears the avatar_url on the profile.

```typescript
import { createError } from 'h3'
import { bearerToken, okJson, readJsonBody, supabaseFetch, supabaseUser, getSupabaseConfig } from '~/server/utils/supabase'

const BUCKET = 'profile-avatars'

export default defineEventHandler(async (event) => {
  const token = bearerToken(event)
  const user = await supabaseUser(event)
  const body = await readJsonBody<{ profileIndex: number }>(event)

  const profileIndex = body?.profileIndex
  if (!profileIndex || profileIndex < 1 || profileIndex > 4) {
    throw createError({ statusCode: 400, statusMessage: 'profileIndex must be 1-4.' })
  }

  const config = getSupabaseConfig()
  const storagePath = `${user.id}/${profileIndex}.jpg`

  // Delete from Supabase Storage. Treat 404 as idempotent success because the
  // user's goal is "no avatar object remains"; fail other non-OK responses
  // before clearing avatar_url so storage failures are not silently hidden.
  const deleteUrl = `${config.url}/storage/v1/object/${BUCKET}/${storagePath}`
  let deleteResponse: Response
  try {
    deleteResponse = await fetch(deleteUrl, {
      method: 'DELETE',
      headers: {
        'Authorization': `Bearer ${config.serviceRoleKey}`
      }
    })
  } catch (e) {
    console.error('[photo.delete] Storage delete request failed:', e)
    throw createError({ statusCode: 502, statusMessage: 'Failed to delete photo from storage.' })
  }

  if (!deleteResponse.ok && deleteResponse.status !== 404) {
    const errorText = await deleteResponse.text()
    console.error('[photo.delete] Storage delete failed:', deleteResponse.status, errorText)
    throw createError({ statusCode: 502, statusMessage: 'Failed to delete photo from storage.' })
  }

  // Clear avatar_url on the profile using the explicit Plan 05-01 contract:
  // p_avatar_url: null preserves by default, so p_clear_avatar: true is required to clear.
  const profiles = await supabaseFetch<Array<{ name: string; avatar_color_hex: string }>>(
    `/rest/v1/profiles?profile_index=eq.${profileIndex}&select=name,avatar_color_hex`,
    { method: 'GET' },
    token
  )

  const currentProfile = profiles?.[0]
  if (currentProfile) {
    await supabaseFetch(
      '/rest/v1/rpc/profile_upsert',
      {
        method: 'POST',
        body: JSON.stringify({
          p_profile_index: profileIndex,
          p_name: currentProfile.name,
          p_avatar_color_hex: currentProfile.avatar_color_hex,
          p_avatar_url: null,
          p_clear_avatar: true
        })
      },
      token
    )
  }

  return okJson({ deleted: true, profileIndex })
})
```

  ```typescript
    state.value.photoUploading = true

    try {
      const formData = new FormData()
  ```

```html
import { ref, watch } from 'vue'
import ProfilePhotoUpload from '~/components/portal/ProfilePhotoUpload.vue'
import { useProfileStore } from '~/composables/useProfileStore'

const props = defineProps<{ profile: any }>()
const emit = defineEmits<{ 'name-updated': [] }>()
const profileStore = useProfileStore()

const editingName = ref(props.profile.name)

// Sync if profile changes externally
watch(() => props.profile.name, (newName) => { editingName.value = newName })

async function handleNameSave() {
  const trimmed = editingName.value.trim()
  if (trimmed === props.profile.name || trimmed.length < 1 || trimmed.length > 30) return

  await profileStore.removeProfilePhoto(props.profile.profile_index)
}
```
