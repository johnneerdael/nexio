    internal data class ProfileJson(
        @SerializedName("id") val id: Int,
        @SerializedName("name") val name: String,
        @SerializedName("avatar_color_hex") val avatarColorHex: String,
        @SerializedName("uses_primary_addons") val usesPrimaryAddons: Boolean = false,
        @SerializedName("avatar_id") val avatarId: String? = null,
        @SerializedName("pin_enabled") val pinEnabled: Boolean = false
    ) {
        fun toDomain() = UserProfile(
            id = id, name = name, avatarColorHex = avatarColorHex,
            usesPrimaryAddons = usesPrimaryAddons, avatarId = avatarId, pinEnabled = pinEnabled
        )
        companion object {
            fun fromDomain(p: UserProfile) = ProfileJson(
                id = p.id, name = p.name, avatarColorHex = p.avatarColorHex,
                usesPrimaryAddons = p.usesPrimaryAddons, avatarId = p.avatarId, pinEnabled = p.pinEnabled
            )
        }
    }

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-03 | Tampering | ProfileDataStore JSON persistence | accept | JSON is stored in app-private DataStore file. Corruption is handled via silent fallback to defaultPrimaryProfile() — no crash, no data leak. Physical device access required to tamper. |
| T-01-04 | Denial of Service | ProfileManager.createProfile | mitigate | Max 4 profiles enforced by `profiles.value.size >= 4` guard. ID range limited to 2..4. No unbounded resource allocation possible. |
| T-01-05 | Information Disclosure | Profile names in DataStore files | accept | Profile names (user-chosen strings in Phase 3) stored in app-private storage. No PII beyond names. Android sandbox prevents cross-app read. |
| T-01-06 | Tampering | ProfileManager.deleteProfile file cleanup | accept | File deletion targets only `_p{id}.preferences_pb` pattern in app-private datastore dir. No path traversal risk — suffix is constructed from integer ID (1-4 range). |
</threat_model>
