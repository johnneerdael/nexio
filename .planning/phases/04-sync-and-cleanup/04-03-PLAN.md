    ```kotlin
    private suspend fun retryPendingRemoteCleanup() {
        val prefs = context.getSharedPreferences("profile_cleanup_state", Context.MODE_PRIVATE)
        val pendingIds = prefs.getStringSet("pending_remote_cleanup", emptySet()) ?: emptySet()
        if (pendingIds.isEmpty()) return

        val remaining = pendingIds.toMutableSet()
        for (idStr in pendingIds) {
            val profileId = idStr.toIntOrNull() ?: continue
            try {
                // Call the sync_delete_profile RPC via ProfileSyncService or directly

            } catch (e: Exception) {
                Log.w(TAG, "Remote cleanup retry failed for profile $profileId: ${e.message}")
            }
        }
        prefs.edit().putStringSet("pending_remote_cleanup", remaining).apply()
    }
    ```

    ```kotlin
    private suspend fun deleteProfileRemote(profileId: Int) {
        val pg = postgrest ?: return
        try {
            pg.rpc("sync_delete_profile", buildJsonObject {
                put("p_profile_id", profileId)
            })
            Log.d("ProfileManager", "Remote profile $profileId deleted")
        } catch (e: Exception) {
            Log.w("ProfileManager", "Remote deletion failed for profile $profileId, scheduling retry: ${e.message}")
            // Schedule retry on next app start per D-04
            val prefs = context.getSharedPreferences("profile_cleanup_state", Context.MODE_PRIVATE)
            val pending = prefs.getStringSet("pending_remote_cleanup", emptySet())?.toMutableSet() ?: mutableSetOf()
            pending.add(profileId.toString())
            // Cap at 4 entries (max profiles)
            if (pending.size > 4) pending.retainAll(pending.take(4).toSet())
            prefs.edit().putStringSet("pending_remote_cleanup", pending).apply()
        }
    }
    ```

    ```kotlin
        }
    ```
