fun rescheduleAirTimeAlarmFromSnapshot() {
    val scheduledReemit = rawSnapshotState.value.scheduledReemit
    val nowMs = System.currentTimeMillis()
    if (scheduledReemit.any { entry ->
            val exactMs = entry.tvdbAvailabilityInstantMs
            exactMs != null && exactMs > 0L && exactMs <= nowMs
        }

}

fun pendingTriggerMs(
    firstAiredMs: Long,
    availabilityInstantMs: Long?,
    tmdbAirDate: String?
): Long? {
    if (availabilityInstantMs != null && availabilityInstantMs > 0L) return availabilityInstantMs
    if (firstAiredMs > 0L) return firstAiredMs
    return parseDateToEpochMs(tmdbAirDate?.trim().orEmpty())
}

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Persisted snapshot -> refresh decision | Local stored `scheduledReemit` rows determine whether a provider refresh is started immediately or a future alarm is scheduled |
| Android receiver -> snapshot service | Boot and exact-alarm permission broadcasts can call `rescheduleAirTimeAlarmFromSnapshot()` |
| Time metadata -> scheduling | TVDB exact instants, provider timestamps, and date strings cross into scheduling decisions |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-20 | Tampering | `AirDateGate.pendingTriggerMs` | mitigate | Treat non-positive timestamps and unparsable dates as unknown, preserving existing behavior instead of forcing a fake due trigger |
| T-08-21 | Denial of Service | `ContinueWatchingSnapshotService.handleScheduledReemit` | mitigate | Start at most one refresh branch per restore/reschedule call when any row is due; future rows still use one soonest scheduler target |
| T-08-22 | Integrity | `ContinueWatchingSnapshotService` visible rows | mitigate | Do not move withheld `scheduledReemit` rows into visible `nextUpItems`; visibility changes only after `ensureFresh(force = true)` rebuilds provider state |
